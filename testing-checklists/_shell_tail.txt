const KEY="aajoo-host-qa-v1";
const STATE_FILE="data/state.json";
let state=JSON.parse(localStorage.getItem(KEY)||"{}");
let filter="all";
// localStorage is only a CACHE. The artifact viewer's sandboxing can
// partition or reset it, and a republished document starts blank — answers
// kept only there kept getting lost. The lasting copy lives in the document
// itself: every change is debounced into artifact.publish({[STATE_FILE]:…}),
// which saves a new version of this artifact carrying the answers. On open,
// the served file and the cache are merged per answer by timestamp, and
// anything the cache held that the document lacked is pushed back up.
const cacheState=()=>{ try{ localStorage.setItem(KEY,JSON.stringify(state)); }catch(e){} };
// Keys are the section and check titles, not their positions, so adding a
// row never moves anyone's answers onto the wrong check.
const id=(s,i)=>DATA[s].t+"::"+DATA[s].items[i][0];
// One-time remap of anything still saved under the old positional keys,
// through the ordering those keys were recorded against (LEGACY_KEYS lives
// beside each document's DATA).
(function(){
  let moved=0;
  for(const k of Object.keys(state)){
    if(!k.includes("|")) continue;
    const to=LEGACY_KEYS[k];
    if(to && state[to]===undefined){ state[to]=state[k]; moved++; }
    delete state[k];
  }
  if(moved) cacheState();
})();

/* ── durable saves into the artifact ─────────────────────────── */
let artifactNS=null, artifactReady=false, canPublish=true, saveTimer=null, saving=false, dirty=false;
const chip=document.getElementById("savechip");
const setChip=(txt,tone)=>{ if(chip){ chip.textContent=txt; if(tone){chip.dataset.tone=tone;}else{delete chip.dataset.tone;} } };
const LOCAL_ONLY="Local only — answers stay in this browser";

const publishState=async()=>{
  if(!artifactReady||!canPublish) return;
  if(saving){ dirty=true; return; }
  saving=true; setChip("Saving…");
  try{
    await artifactNS.publish({[STATE_FILE]: JSON.stringify(state)});
    setChip("Saved to doc ✓","ok");
  }catch(err){
    const code=err&&err.code;
    if(code==="not_writer"||code==="not_granted"||code==="not_declared"||code==="capability_disabled"||code==="capability_removed"){
      // Permanent for this view — say so once and fall back to the cache.
      canPublish=false; setChip(LOCAL_ONLY,"warn");
    }else if(code==="conflict"){
      // Another view published first; the shell reloads to the winner and
      // the merge on the next load reconciles. Nothing else to do here.
      setChip("Syncing…");
    }else if(code==="rate_limited"){
      dirty=true; setTimeout(()=>{ if(dirty){dirty=false; publishState();} },15000);
    }else{
      setChip("Save failed — kept in this browser, retrying…","warn");
      dirty=true; setTimeout(()=>{ if(dirty){dirty=false; publishState();} },8000);
    }
  }finally{
    saving=false;
    if(dirty&&canPublish){ dirty=false; publishState(); }
  }
};
const save=()=>{
  cacheState();
  if(!artifactReady||!canPublish) return;
  setChip("Saving…");
  clearTimeout(saveTimer);
  saveTimer=setTimeout(publishState,2000);
};
// Per-answer newest-wins, honouring the reset tombstone (__cleared): a Reset
// must not be undone by an older cache on another device.
const mergeStates=(server,local)=>{
  const cleared=Math.max(Number(server.__cleared)||0, Number(local.__cleared)||0);
  const out={};
  if(cleared) out.__cleared=cleared;
  let localAhead=(Number(local.__cleared)||0)>(Number(server.__cleared)||0);
  const keys=new Set([...Object.keys(server),...Object.keys(local)]);
  for(const k of keys){
    if(k==="__cleared") continue;
    const s=server[k]||null, l=local[k]||null;
    const st=s?(Number(s.t)||0):-1, lt=l?(Number(l.t)||0):-1;
    const useLocal=lt>st;
    const win=useLocal?l:s;
    if(!win) continue;
    if(Math.max(useLocal?lt:st,0)<cleared) continue;
    out[k]=win;
    if(useLocal) localAhead=true;
  }
  return {merged:out, localAhead};
};
(async()=>{
  try{
    if(!(window.claude&&window.claude.use)){ setChip(LOCAL_ONLY,"warn"); return; }
    artifactNS=await window.claude.use("artifact");
    if(!artifactNS){ setChip(LOCAL_ONLY,"warn"); return; }
    let server={};
    try{
      const res=await fetch(STATE_FILE,{cache:"no-store"});
      if(res.ok) server=(await res.json())||{};
    }catch(e){ /* no saved file yet — first run */ }
    const {merged, localAhead}=mergeStates(server,state);
    state=merged; cacheState(); artifactReady=true;
    render();
    if(localAhead){ publishState(); } else { setChip("Saved to doc ✓","ok"); }
  }catch(e){ setChip(LOCAL_ONLY,"warn"); }
})();
window.addEventListener("pagehide",()=>{ if(saveTimer){ clearTimeout(saveTimer); publishState(); } });

const toast=(m)=>{const t=document.getElementById("toast");t.textContent=m;t.classList.add("on");clearTimeout(t._h);t._h=setTimeout(()=>t.classList.remove("on"),1900);};

function render(){
  const out=document.getElementById("out"), rail=document.getElementById("rail");
  out.innerHTML=""; rail.innerHTML="";
  let total=0,pass=0,fail=0,block=0;

  DATA.forEach((sec,si)=>{
    let sDone=0;
    const section=document.createElement("section");
    section.id="s"+si;
    const rows=document.createElement("div"); rows.className="rows";

    sec.items.forEach((it,ii)=>{
      const [title,exp,e1,e2,expect]=it;
      const k=id(si,ii); const st=state[k]||{};
      total++;
      if(st.v==="pass")pass++; else if(st.v==="fail")fail++; else if(st.v==="block")block++;
      if(st.v)sDone++;
      if(filter==="todo"&&st.v) return;
      if(filter==="bad"&&st.v!=="fail"&&st.v!=="block") return;

      const row=document.createElement("div");
      row.className="row"; if(st.v)row.dataset.s=st.v;
      const eps=[e1,e2].filter(Boolean).map(e=>`<span class="ep">${e}</span>`).join("");
      row.innerHTML=`
        <div class="row-top">
          <div class="what">
            <div class="title">${title}</div>
            <div class="exp">${exp}</div>
            <div class="exp" style="margin-top:5px"><strong style="color:var(--ink)">Expect:</strong> ${expect}</div>
            ${eps?`<div class="eps">${eps}</div>`:""}
          </div>
          <div class="seg" role="group" aria-label="Result for ${title}">
            <button data-v="pass" aria-pressed="${st.v==="pass"}">Pass</button>
            <button data-v="fail" aria-pressed="${st.v==="fail"}">Fail</button>
            <button data-v="block" aria-pressed="${st.v==="block"}">Blocked</button>
          </div>
        </div>
        <textarea class="note ${st.v==="fail"||st.v==="block"||st.n?"":"hide"}"
          placeholder="What did you see? Steps, message, anything odd.">${st.n||""}</textarea>`;

      row.querySelectorAll(".seg button").forEach(b=>{
        b.onclick=()=>{
          const v=b.dataset.v;
          state[k]={...(state[k]||{}), v: st.v===v? null : v, t: Date.now()};
          if(!state[k].v && !state[k].n) delete state[k];
          save(); render();
        };
      });
      const ta=row.querySelector(".note");
      ta.oninput=()=>{ state[k]={...(state[k]||{}), n:ta.value, t: Date.now()}; if(!state[k].v&&!state[k].n)delete state[k]; save(); updateBar(); };
      rows.appendChild(row);
    });

    section.innerHTML=`<div class="head"><span class="num">${String(si+1).padStart(2,"0")}</span><h2>${sec.t}</h2></div><p class="sub">${sec.d}</p>`;
    section.appendChild(rows);
    if(rows.children.length) out.appendChild(section);

    const a=document.createElement("a");
    a.href="#s"+si;
    a.className=sDone===sec.items.length?"done":"";
    a.innerHTML=`<b class="tnum">${String(sDone).padStart(2,"0")}/${String(sec.items.length).padStart(2,"0")}</b><span>${sec.t}</span>`;
    rail.appendChild(a);
  });

  window._tot={total,pass,fail,block};
  updateBar();
}

function updateBar(){
  const {total,pass,fail,block}=window._tot;
  document.getElementById("bp").style.width=(pass/total*100)+"%";
  document.getElementById("bf").style.width=(fail/total*100)+"%";
  document.getElementById("bb").style.width=(block/total*100)+"%";
  document.getElementById("cnt").textContent=`${pass+fail+block} / ${total} · ${fail} failed · ${block} blocked`;
}

document.getElementById("filter").onclick=(e)=>{
  filter = filter==="all" ? "todo" : filter==="todo" ? "bad" : "all";
  e.target.textContent = "Show: " + (filter==="all"?"all":filter==="todo"?"not done":"problems");
  render();
};

document.getElementById("copy").onclick=()=>{
  let out="# Aajoo host testing — findings\n\n";
  let n=0;
  DATA.forEach((sec,si)=>{
    const bad=sec.items.map((it,ii)=>({it,st:state[id(si,ii)]||{}}))
      .filter(x=>x.st.v==="fail"||x.st.v==="block");
    if(!bad.length) return;
    out+=`## ${sec.t}\n\n`;
    bad.forEach(({it,st})=>{
      n++;
      out+=`**${st.v==="fail"?"FAILED":"BLOCKED"} — ${it[0]}**\n`;
      out+=`- Expected: ${it[4]}\n`;
      if(it[2]||it[3]) out+=`- Endpoints: ${[it[2],it[3]].filter(Boolean).join(", ")}\n`;
      out+=`- What happened: ${st.n?st.n:"(no note)"}\n\n`;
    });
  });
  const {total,pass,fail,block}=window._tot;
  out = `Tested ${pass+fail+block} of ${total} — ${pass} passed, ${fail} failed, ${block} blocked.\n\n` + (n?out:"No failures recorded.\n");

  // Then the whole run, one line per answered check. The failures above are
  // what needs fixing; this is the record that the rest were actually tested,
  // which is the half a Claude session cannot obtain any other way — this
  // document's saved answers are private to your signed-in browser.
  const MARK={pass:"PASS ",fail:"FAIL ",block:"BLOCK"};
  let done="";
  DATA.forEach((sec,si)=>{
    const marked=sec.items.map((it,ii)=>({it,st:state[id(si,ii)]||{}})).filter(x=>x.st.v);
    if(!marked.length) return;
    done+=`\n### ${sec.t}\n`;
    marked.forEach(({it,st})=>{ done+=`${MARK[st.v]||st.v} — ${it[0]}${st.n?`  · ${st.n.replace(/\s+/g," ").slice(0,160)}`:""}\n`; });
  });
  const untested=total-(pass+fail+block);
  out += `\n---\n\n## Full run\n${done||"\nNothing marked yet.\n"}`;
  if(untested) out += `\n${untested} check${untested>1?"s":""} not marked.\n`;

  navigator.clipboard.writeText(out).then(
    ()=>toast(n?`Copied ${n} problem${n>1?"s":""} + the full run — paste it to Claude`:"Copied the full run — paste it to Claude"),
    ()=>toast("Couldn't copy — select the page and copy manually")
  );
};

document.getElementById("reset").onclick=()=>{
  if(!confirm("Clear every answer and note? This cannot be undone.")) return;
  // The tombstone timestamp stops an older cache on another device from
  // quietly resurrecting everything the reset removed.
  state={__cleared: Date.now()}; save(); render(); toast("Checklist cleared");
};

render();
</script>
